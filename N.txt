        function addPledge(event) {
            event.preventDefault();
            const nameInput = document.getElementById('sender-name');
            const msgInput = document.getElementById('sender-message');
            const wall = document.getElementById('messages-wall');

            const name = nameInput.value.trim();
            const msg = msgInput.value.trim();
            if (!name || !msg) return;

            const cardColors = [
                'bg-[#fff6f9] border-pink-200',
                'bg-[#f6eeff] border-purple-200',
                'bg-[#eefaf3] border-emerald-200',
                'bg-[#eef7fd] border-sky-200',
                'bg-[#fef9ea] border-amber-200'
            ];
            const randomStyle = cardColors[Math.floor(Math.random() * cardColors.length)];

            const card = document.createElement('div');
            card.className = `${randomStyle} rounded-2xl p-4 border-2 shadow-sm transform transition-all duration-500 scale-95 opacity-0`;
            
            const header = document.createElement('div');
            header.className = 'flex items-center justify-between mb-1.5';
            
            const nameSpan = document.createElement('span');
            nameSpan.className = 'font-extrabold text-pastel-text text-sm';
            nameSpan.textContent = '💌 ' + name;
            
            const iconSpan = document.createElement('span');
            iconSpan.textContent = '🕊️🌸';
            
            header.appendChild(nameSpan);
            header.appendChild(iconSpan);

            const bodyP = document.createElement('p');
            bodyP.className = 'text-[#6e3b52] text-sm font-medium';
            bodyP.textContent = msg;

            card.appendChild(header);
            card.appendChild(bodyP);

            wall.insertBefore(card, wall.firstChild);
            requestAnimationFrame(() => {
                card.classList.remove('scale-95', 'opacity-0');
            });

            nameInput.value = '';
            msgInput.value = '';
            incrementCounter();
            burstHearts();
            showToast('شكراً لك يا ' + name + '! حلّق ظرف تسامحك في سماء المبادرة 💌🕊️');
        }
